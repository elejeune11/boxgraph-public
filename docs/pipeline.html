Commands

python -m pipeline.add paper.pdf --id smith2015 [--pages 1-12] [--note "…"] [--relate-with bell1978]
python -m pipeline.attach code.zip --id smith_code --kind software --papers smith2015 # or a data file with --kind dataset
python -m pipeline.run_papers --plan # what would run, with budget caps
python -m pipeline.run_papers [--papers smith2015] [--concurrency 3]
python -m pipeline.run_papers --status # state of every job on this machine
python -m pipeline.run_papers --reset read:smith2015 # release a parked job after looking at it
